The markup below is the theme's. */ -}} {{- $navRoot := cond (eq site.Home.Type "docs") site.Home .FirstSection -}} {{- $pages := slice $navRoot | append (partial "custom/sidebar-order.html" $navRoot) -}} {{- $prev := false -}} {{- $next := false -}} {{- range $i, $p := $pages -}} {{- if eq $p.RelPermalink $.RelPermalink -}} {{- if gt $i 0 }}{{ $prev = index $pages (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $pages) }}{{ $next = index $pages (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{- if eq .Params.prev false }} {{- $prev = false -}} {{ else }} {{- with .Params.prev -}} {{- with $.Site.GetPage . }}{{ $prev = . }}{{ end -}} {{- end -}} {{- end -}} {{- if eq .Params.next false }} {{- $next = false -}} {{ else }} {{- with .Params.next -}} {{- with $.Site.GetPage . }}{{ $next = . }}{{ end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
