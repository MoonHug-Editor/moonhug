The pager walks this list. */ -}} {{- $out := slice -}} {{- $items := where (union .RegularPages .Sections) "Params.sidebar.exclude" "!=" true -}} {{- range $items.ByWeight -}} {{- if not .Params.sidebar.separator -}} {{- $out = $out | append . -}} {{- end -}} {{- if .IsSection -}} {{- $out = $out | append (partial "custom/sidebar-order.html" .) -}} {{- end -}} {{- end -}} {{- return $out -}}
