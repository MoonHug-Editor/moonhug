A relative link that climbs above the content root can only mean the repo root, so its leading ../ are dropped first. Everything else in this file is the theme's hook, unchanged. */ -}} {{- $pagePath = path.Clean $pagePath -}} {{- $pagePath = replaceRE `^/+` "" $pagePath -}} {{- $pagePath = replaceRE `^(\.\./)+` "" $pagePath -}} {{- /* A page says where it sits (its `path:` front matter), and `mh docs` writes the repo path -> site path table to data/pages.json. */ -}} {{- with index site.Data.pages (printf "%s.md" $pagePath) -}} {{- $pagePath = . -}} {{- else -}} {{- $pagePath = replaceRE `^docs/` "" $pagePath -}} {{- $pagePath = replaceRE `^plugins/([^/]+)/docs/` "plugins/$1/" $pagePath -}} {{- end -}} {{- /* The README is the home page. */ -}} {{- $pagePath = replaceRE `^README$` "" $pagePath -}} {{- /* Looked up from the site root: after the join above the path is root-relative, and a relative lookup would try the page's own folder first. */ -}} {{- with .PageInner.GetPage (printf "/%s" $pagePath) -}} {{- $dest = printf "%s%s%s" .RelPermalink $query $fragment -}} {{- $resolvedPage = true -}} {{- end -}} {{- end -}} {{- if and $parsedURL (not $resolvedPage) $dest (hasPrefix $dest "/") -}} {{- with or (.PageInner.GetPage $url.Path) (.PageInner.Resources.Get $url.Path) (resources.Get $url.Path) -}} {{- $dest = printf "%s%s%s" .RelPermalink $query $fragment -}} {{- else -}} {{- $hasBasePrefix := and (ne site.Home.RelPermalink "/") (hasPrefix $dest site.Home.RelPermalink) -}} {{- if not $hasBasePrefix -}} {{- $dest = (relURL (strings.TrimPrefix "/" $dest)) -}} {{- end -}} {{- end -}} {{- end -}} {{- with . -}} {{- $isExternal := strings.HasPrefix .Destination "http" -}} {{- .Text | safeHTML -}} {{- if and .Page.Site.Params.externalLinkDecoration $isExternal -}} {{- partial "utils/icon.html" (dict "name" "arrow-up-right" "attributes" `class="hx:inline hx:rtl:rotate-270 hx:align-baseline" height="1em" aria-hidden="true"`) -}} {{- end -}} {{- end -}}
